<script>
/* Navigation, demo estimate form, deep links, before/after slider, project filters, reveal. No dependencies. */
(function () {
  'use strict';
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var $$ = function (s, r) { return Array.prototype.slice.call((r || document).querySelectorAll(s)); };

  /* ---- header shadow on scroll ---- */
  var header = $('.site-header');
  if (header) {
    var onScroll = function () { header.classList.toggle('is-scrolled', window.scrollY > 8); };
    window.addEventListener('scroll', onScroll, { passive: true }); onScroll();
  }

  /* ---- mobile menu ---- */
  var btn = $('.menu-btn'), nav = $('#site-nav');
  if (btn && nav) {
    var setMenu = function (open) { nav.classList.toggle('is-open', open); btn.setAttribute('aria-expanded', open ? 'true' : 'false'); };
    btn.addEventListener('click', function () { setMenu(!nav.classList.contains('is-open')); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && nav.classList.contains('is-open')) { setMenu(false); btn.focus(); } });
    nav.addEventListener('click', function (e) { if (e.target.closest('a')) setMenu(false); });
  }

  /* ---- estimate form (demo) ---- */
  var form = document.getElementById('estimate-form');

  function applyHash() {
    var m = /^#estimate-([a-z0-9-]+)$/.exec(window.location.hash);
    if (!m) return;
    var card = document.getElementById('estimate');
    if (form && form.elements.message && !form.elements.message.value) {
      var pick = $('.pick[href="#estimate-' + m[1] + '"] strong');
      var t = pick ? pick.textContent.trim() : '';
      if (t) { form.elements.message.value = m[1] === 'unsure' ? "I'm not sure yet what I need. " : 'I need: ' + t + '. '; clearErr('message'); }
    }
    if (card) {
      card.scrollIntoView({ behavior: window.matchMedia('(prefers-reduced-motion: reduce)').matches ? 'auto' : 'smooth', block: 'start' });
      card.classList.remove('pulse'); void card.offsetWidth; card.classList.add('pulse');
    }
  }

  var rules = {
    full_name: function (v) { return v.trim().length >= 2 ? '' : 'Please enter your name.'; },
    phone: function (v) { return v.replace(/\D/g, '').length === 10 ? '' : 'Enter a 10-digit phone number so we can reach you.'; },
    message: function (v) { return v.trim().length >= 5 ? '' : 'Please tell us briefly what you need done.'; }
  };
  var idFor = { full_name: 'f-name', phone: 'f-phone', message: 'f-message' };

  function valueOf(n) { var e = form.elements[n]; return (e && e.value) || ''; }
  function boxOf(n) { return form.elements[n].closest('.field'); }
  function clearErr(n) {
    if (!form || !form.elements[n]) return;
    var err = document.getElementById(idFor[n] + '-err'); if (err) err.hidden = true;
    boxOf(n).classList.remove('has-err');
  }
  function check(n) {
    var msg = rules[n](valueOf(n)), err = document.getElementById(idFor[n] + '-err'), box = boxOf(n);
    box.classList.toggle('has-err', !!msg);
    var el = form.elements[n]; el.setAttribute('aria-invalid', msg ? 'true' : 'false'); if (msg) el.setAttribute('aria-describedby', err.id); else el.removeAttribute('aria-describedby');
    if (msg) { err.textContent = msg; err.hidden = false; } else { err.hidden = true; }
    return !msg;
  }
  function focusField(n) { var e = form.elements[n]; (e.length ? e[0] : e).focus(); }
  function esc(s) { return String(s).replace(/[&<>"']/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]; }); }

  if (form && form.getAttribute('data-demo') === 'true') {
    var phone = form.elements.phone;
    phone.addEventListener('input', function () {
      var d = phone.value.replace(/\D/g, '');
      if (d.length > 10 && d.charAt(0) === '1') d = d.slice(1);
      d = d.slice(0, 10);
      var out = d;
      if (d.length > 6) out = '(' + d.slice(0, 3) + ') ' + d.slice(3, 6) + '-' + d.slice(6);
      else if (d.length > 3) out = '(' + d.slice(0, 3) + ') ' + d.slice(3);
      phone.value = out;
    });
    ['full_name', 'phone', 'message'].forEach(function (n) {
      form.elements[n].addEventListener('blur', function () { if (form.elements[n].value) check(n); });
      form.elements[n].addEventListener('input', function () { if (form.elements[n].getAttribute('aria-invalid') === 'true') check(n); });
    });

    form.addEventListener('submit', function (e) {
      e.preventDefault();
      var firstBad = null;
      Object.keys(rules).forEach(function (n) { if (!check(n) && !firstBad) firstBad = n; });
      if (firstBad) { focusField(firstBad); document.getElementById('form-status').textContent = 'Please fix the highlighted fields.'; return; }
      var data = { name: valueOf('full_name').trim(), phone: valueOf('phone'), message: valueOf('message').trim() };
      var call = $('.topbar-phone, .icon-call'), href = call ? call.getAttribute('href') : '#';
      var disp = ($('.hero-call strong') || $('.topbar-phone') || {}).textContent || '';
      var card = form.closest('.form-card'), title = $('.form-title', card), sub = $('.form-sub', card);
      if (title) title.textContent = 'Thanks, ' + data.name.split(' ')[0] + '!';
      if (sub) sub.remove();
      form.outerHTML =
        '<div class="success" tabindex="-1" id="form-success">' +
        '<span class="demo-flag">Demo mode: nothing was sent</span>' +
        '<h3>Here is what would happen next</h3>' +
        '<dl><dt>Name</dt><dd>' + esc(data.name) + '</dd><dt>Phone</dt><dd>' + esc(data.phone) + '</dd>' +
        '<dt>Service needed</dt><dd>' + esc(data.message) + '</dd></dl>' +
        '<ol><li>We call or text you to confirm the details.</li><li>We book a time to inspect and measure.</li><li>You receive a written estimate. No obligation.</li></ol>' +
        '<p class="fine">This preview form does not store or send anything. Connect a GoHighLevel form to receive real requests.</p>' +
        '<a class="btn btn-call btn-block" href="' + esc(href) + '"><svg class="ico" aria-hidden="true"><use href="#i-phone"/></svg>Urgent? Call <span class="nowrap">' + esc(disp.trim()) + '</span></a></div>';
      var ok = document.getElementById('form-success'); if (ok) ok.focus();
    });
  }
  window.addEventListener('hashchange', applyHash);
  applyHash();

  /* ---- before / after slider ---- */
  $$('[data-ba]').forEach(function (ba) {
    var stage = $('.ba-stage', ba), range = $('.ba-range', ba);
    var before = $('#ba-before', ba), after = $('#ba-after', ba), cap = $('#ba-cap-text', ba);
    function setPos(v) { stage.style.setProperty('--p', v + '%'); }
    range.addEventListener('input', function () { setPos(range.value); });
    setPos(range.value);
    $$('[data-ba-tab]', ba).forEach(function (t) {
      t.addEventListener('click', function () {
        var g = function (k) { return t.getAttribute('data-' + k); };
        $$('[data-ba-tab]', ba).forEach(function (o) { var on = o === t; o.classList.toggle('is-on', on); o.setAttribute('aria-pressed', on ? 'true' : 'false'); });
        before.src = g('before'); after.src = g('after');
        before.alt = g('alt-before'); after.alt = g('alt-after'); cap.textContent = g('cap');
        range.value = 50; setPos(50);
      });
    });
  });

  /* ---- project filters ---- */
  var filters = $$('.filter[data-filter]');
  if (filters.length) {
    var cards = $$('#project-grid .card'), status = $('#filter-status'), empty = $('#filter-empty');
    filters.forEach(function (f) {
      f.addEventListener('click', function () {
        var key = f.getAttribute('data-filter'), shown = 0;
        filters.forEach(function (o) { var on = o === f; o.classList.toggle('is-on', on); o.setAttribute('aria-pressed', on ? 'true' : 'false'); });
        cards.forEach(function (c) { var show = key === 'all' || c.getAttribute('data-cat') === key; c.hidden = !show; if (show) shown++; });
        if (empty) empty.hidden = shown > 0;
        if (status) status.textContent = 'Showing ' + shown + (shown === 1 ? ' project' : ' projects');
      });
    });
  }


  /* ---- photo gallery lightbox ---- */
  var tiles = $$('[data-lightbox]');
  if (tiles.length) {
    var box = document.createElement('div'); box.className = 'lb'; box.setAttribute('role', 'dialog'); box.setAttribute('aria-modal', 'true'); box.setAttribute('aria-label', 'Photo viewer');
    box.innerHTML = '<button type="button" aria-label="Close photo">&times;</button><img alt=""><p></p>';
    document.body.appendChild(box);
    var lbImg = $('img', box), lbCap = $('p', box), lbClose = $('button', box), lastFocus = null;
    var closeLb = function () { box.classList.remove('on'); if (lastFocus) lastFocus.focus(); };
    tiles.forEach(function (t) {
      t.addEventListener('click', function () {
        lastFocus = t; lbImg.src = t.getAttribute('data-lightbox'); lbImg.alt = (t.querySelector('img') || {}).alt || ''; lbCap.textContent = t.getAttribute('data-cap') || '';
        box.classList.add('on'); lbClose.focus();
      });
    });
    lbClose.addEventListener('click', closeLb);
    box.addEventListener('click', function (e) { if (e.target === box) closeLb(); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape' && box.classList.contains('on')) closeLb(); });
  }

  /* ---- gentle reveal on scroll (skipped for reduced motion / no IntersectionObserver) ---- */
  if (!window.__noReveal && 'IntersectionObserver' in window && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('in'); io.unobserve(en.target); } });
    }, { rootMargin: '0px 0px -8% 0px', threshold: 0.05 });
    $$('.section-head, .card, .steps li, .pick, .callout, .compare > div, .checklist li, .faq details, .ba').forEach(function (el) {
      var r = el.getBoundingClientRect();
      if (r.top < window.innerHeight) return;
      el.classList.add('rv'); io.observe(el);
    });
  }
})();

</script>
